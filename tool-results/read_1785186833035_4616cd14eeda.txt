   214→function useTileStatus(mapRef: React.RefObject<LeafletMap | null>) {
   215→  const [loading, setLoading] = useState(false);
   216→  const [error, setError] = useState(false);
   217→  const clearTimeoutRef = useRef<ReturnType<typeof setTimeout> | undefined>(undefined);
   218→
   219→  useEffect(() => {
   220→    const map = mapRef.current;
   221→    if (!map) return;
   222→
   223→    const onLoading = () => {
   224→      if (clearTimeoutRef.current) clearTimeout(clearTimeoutRef.current);
   225→      setError(false);
   226→      setLoading(true);
   227→    };
   228→    const onLoad = () => {
   229→      setLoading(false);
   230→      setError(false);
   231→    };
   232→    const onTileError = () => {
   233→      setLoading(false);
   234→      setError(true);
   235→      clearTimeoutRef.current = setTimeout(() => setError(false), 5000);
   236→    };
   237→
   238→    map.on('tileloadstart', onLoading);
   239→    map.on('tileload', onLoad);
   240→    map.on('load', onLoad);
   241→    map.on('tileerror', onTileError);
   242→
   243→    return () => {
   244→      map.off('tileloadstart', onLoading);
   245→      map.off('tileload', onLoad);
   246→      map.off('load', onLoad);
   247→      map.off('tileerror', onTileError);
   248→      if (clearTimeoutRef.current) clearTimeout(clearTimeoutRef.current);
   249→    };
   250→  }, [mapRef]);
   251→
   252→  return { loading, error };
   253→}

   460→                        <div><b>সংখ্যা:</b> {toBnNum(p.totalQuantity)} টি</div>
   461→                        {p.plantingDate && <div><b>রোপণ তারিখ:</b> {p.plantingDate}</div>}
   462→                        {p.farmerName && (
   463→                          <div><b>কৃষক:</b> {p.farmerName}{p.farmerMobile ? ` (${p.farmerMobile})` : ''}</div>
   464→                        )}
   465→                        {p.saaoName && <div><b>SAAO:</b> {p.saaoName}</div>}
   466→                        {p.officerName && <div><b>মনিটরিং অফিসার:</b> {p.officerName}</div>}
   467→                        <div className="font-mono text-[10px] text-slate-500 mt-1">
   468→                          {p.latitude.toFixed(5)}, {p.longitude.toFixed(5)}

   471→                      </div>
   472→                    </div>
   473→                  </Popup>
   474→                </CircleMarker>
   475→              );
   476→            })
   477→          : SEED_PLANTATIONS
   478→              .filter((p) => p.latitude !== 0 && p.longitude !== 0)
   479→              .map((p) => (

   481→                  key={`seed-${p.sl}`}
   482→                  center={[p.latitude, p.longitude]}
   483→                  radius={6}

   495→                        {toBnNum(p.count)} টি · {p.district} / {p.upazila}
   496→                      </div>
   497→                    </div>
   498→                  </Tooltip>
   499→                  <Popup>
   500→                    <div className="text-xs min-w-[180px]">
   501→                      <div className="font-bold text-emerald-800 mb-1 flex items-center gap-1">
   502→                        <Trees size={12} /> {p.speciesName}
   503→                      </div>
   504→                      <div className="space-y-0.5 text-[11px] text-slate-700">
   505→                        <div><b>জেলা:</b> {p.district} · {p.upazila}</div>
   506→                        <div><b>সংখ্যা:</b> {toBnNum(p.count)} টি</div>
   507→                        <div><b>রোপণ তারিখ:</b> {p.plantingDate}</div>
   508→                        <div><b>পরিচর্যাকারী:</b> {p.caretaker}</div>
   509→                        <div className="font-mono text-[10px] text-slate-500 mt-1">
   510→                          {p.latitude.toFixed(5)}, {p.longitude.toFixed(5)}
   511→                        </div>
   512→                        <div className="text-[10px] text-amber-600 mt-1">📦 অফলাইন সিড স্ন্যাপশট</div>
   513→                      </div>
   514→                    </div>
   515→                  </Popup>
   516→                </CircleMarker>
   517→              ))}
   518→
   519→        {/* Real plantation submissions, color-coded per upazila. Click a
   520→            marker to see the summary, then "সম্পাদনা" opens MapEditModal
   521→            for on-site corrections (location, seedling counts, caretaker
   522→            contact) — writes back through the same offline sync queue
   523→            used by the entry form. */}
   524→        {filteredSubmissions.map((s) => {
   525→          const color = colorForUpazila(s.upazila);
   526→          const totalCount = s.seedlings.reduce((sum, sd) => sum + (sd.count || 0), 0);
   527→          return (
   528→            <CircleMarker
   529→              key={s.id}
   530→              center={[s.latitude, s.longitude]}
   531→              radius={7}
   532→              pathOptions={{
   533→                color,
   534→                fillColor: color,
   535→                fillOpacity: 0.8,
   536→                weight: 2,
   537→              }}
   538→            >
   539→              <Tooltip direction="top" offset={[0, -6]} opacity={1}>
   540→                <div className="text-[10px] leading-tight">
   541→                  <div className="font-bold">{s.village}</div>
   542→                  <div className="text-slate-600">
   543→                    {toBnNum(totalCount)} টি চারা · {s.upazila}

   545→                </div>
   546→              </Tooltip>
   547→              <Popup>
   548→                <div className="text-xs min-w-[180px] space-y-1">
   549→                  <div className="font-bold text-emerald-800 flex items-center gap-1">

   551→                  </div>
   552→                  <div className="space-y-0.5 text-[11px] text-slate-700">
   553→                    <div><b>উপজেলা:</b> {s.upazila} · {s.union}</div>
   554→                    <div><b>মোট চারা:</b> {toBnNum(totalCount)} টি</div>

   556→                    <div className="font-mono text-[10px] text-slate-500">
   557→                      {s.latitude.toFixed(5)}, {s.longitude.toFixed(5)}

   560→                  </div>
   561→                  <button
   562→                    onClick={() => setEditingSubmission(s)}
   563→                    className="w-full mt-1 py-1.5 rounded-lg bg-emerald-700 text-white text-[11px] font-semibold hover:bg-emerald-800 cursor-pointer"
   564→                  >
   565→                    সম্পাদনা
   566→                  </button>
   567→                </div>
   568→              </Popup>
   569→            </CircleMarker>
   570→          );
   571→        })}
   572→
   573→        <BoundsTracker onBoundsChange={setBounds} />
   574→      </MapContainer>
   575→
   576→      <LayerSwitcher active={activeLayer} onChange={setActiveLayer} />
   577→      <MapFilterBar
   578→        query={searchQuery}
   579→        onQueryChange={setSearchQuery}
   580→        activeUpazilas={activeUpazilas}
   581→        onToggleUpazila={toggleUpazila}
   582→        onClearUpazilas={() => setActiveUpazilas([])}
   583→        resultCount={filteredSubmissions.length}
   584→      />
   585→      <NDVILegend visible={showLegend} />
   586→      <CustomZoomControl mapRef={mapRef} />
   587→      <BoundsOverlay bounds={bounds} />
   588→      <TileStatusIndicator loading={tileLoading} error={tileError} />
   589→
   590→      {/* Existing compact cloud-pipeline FAB */}
   591→      <div className="absolute bottom-3 right-2 sm:bottom-4 sm:right-3 z-[1000]">
   592→        <CloudPipelineButton state={pipelineState} onRun={runPipeline} />
   593→      </div>
   594→
   595→      {/* NDVI Simulator & Canopy Growth Tracker — opens a side panel */}
   596→      <div className="absolute bottom-3 left-2 sm:bottom-4 sm:left-3 z-[1000]">
   597→        <button
   598→          onClick={() => setSimulatorOpen(true)}
   599→          className={`flex items-center gap-1.5 px-3 py-2 rounded-full shadow-lg transition-all active:scale-95 cursor-pointer text-xs font-bold ${
   600→            simulatorOpen
   601→              ? 'bg-emerald-800 text-white'
   602→              : 'bg-white/95 backdrop-blur text-emerald-800 hover:bg-emerald-50 border border-emerald-200'
   603→          }`}
   604→          title="উপগ্রহ এনডিভিআই সিমুলেটর ও বৃদ্ধি ট্র্যাকিং"
   605→        >
   606→          <Satellite size={14} className={simulatorOpen ? 'text-emerald-300' : 'text-emerald-600'} />
   607→          <span className="hidden sm:inline">NDVI সিমুলেটর</span>
   608→          <span className="sm:hidden">NDVI</span>
   609→        </button>
   610→      </div>
   611→
   612→      {result && pipelineState !== 'running' && (
   613→        <ResultOverlay result={result} onClose={() => setResult(null)} />
   614→      )}
   615→
   616→      {/* The full NDVI + Canopy Growth Tracker panel.
   617→          Shares pipelineState + runPipeline so the simulator's "Run Cloud
   618→          Pipeline" button and the existing compact FAB stay in sync. */}
   619→      <NDVISimulatorPanel
   620→        open={simulatorOpen}
   621→        onClose={() => setSimulatorOpen(false)}
   622→        pipelineState={pipelineState}
   623→        onRunPipeline={runPipeline}
   624→        liveNdvi={result?.ndvi_mean ?? null}
   625→      />
   626→
   627→      {editingSubmission && (
   628→        <MapEditModal
   629→          submission={editingSubmission}
   630→          onSave={handleSaveEdit}
   631→          onClose={() => setEditingSubmission(null)}
   632→        />
   633→      )}
   634→    </div>
   635→  );
