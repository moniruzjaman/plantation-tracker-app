    60→const DEFAULT_SPECIES_COLOR = '#047857'; // emerald-700, preserves existing palette for unknowns

    62→/** Infer a color from speciesName by checking for known keywords */
    63→function colorForSpecies(speciesName: string): string {
    64→  if (!speciesName) return DEFAULT_SPECIES_COLOR;
    65→  // Check each category keyword against the species name
    66→  for (const [keyword, color] of Object.entries(SPECIES_CATEGORY_COLORS)) {
    67→    if (speciesName.includes(keyword)) return color;
    68→  }
    69→  return DEFAULT_SPECIES_COLOR;
    70→}

    75→  if (upazila && UPAZILA_COLORS[upazila]) return UPAZILA_COLORS[upazila];
    76→  return colorForSpecies(speciesName);

   270→function useTileStatus(mapRef: React.RefObject<LeafletMap | null>) {
   271→  const [loading, setLoading] = useState(false);
   272→  const [error, setError] = useState(false);
   273→  const clearTimeoutRef = useRef<ReturnType<typeof setTimeout> | undefined>(undefined);
   274→
   275→  useEffect(() => {
   276→    const map = mapRef.current;
   277→    if (!map) return;
   278→
   279→    const onLoading = () => {
   280→      if (clearTimeoutRef.current) clearTimeout(clearTimeoutRef.current);
   281→      setError(false);
   282→      setLoading(true);
   283→    };
   284→    const onLoad = () => {
   285→      setLoading(false);
   286→      setError(false);
   287→    };
   288→    const onTileError = () => {
   289→      setLoading(false);
   290→      setError(true);
   291→      clearTimeoutRef.current = setTimeout(() => setError(false), 5000);
   292→    };
   293→
   294→    map.on('tileloadstart', onLoading);
   295→    map.on('tileload', onLoad);
   296→    map.on('load', onLoad);
   297→    map.on('tileerror', onTileError);
   298→
   299→    return () => {
   300→      map.off('tileloadstart', onLoading);
   301→      map.off('tileload', onLoad);
   302→      map.off('load', onLoad);
   303→      map.off('tileerror', onTileError);
   304→      if (clearTimeoutRef.current) clearTimeout(clearTimeoutRef.current);
   305→    };
   306→  }, [mapRef]);
   307→
   308→  return { loading, error };
   309→}
