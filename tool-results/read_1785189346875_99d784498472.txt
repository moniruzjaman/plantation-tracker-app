   270→   270→   270→function useTileStatus(mapRef: React.RefObject<LeafletMap | null>) {
   271→   271→   271→  const [loading, setLoading] = useState(false);
   272→   272→   272→  const [error, setError] = useState(false);
   273→   273→   273→  const clearTimeoutRef = useRef<ReturnType<typeof setTimeout> | undefined>(undefined);
   274→   274→   274→
   275→   275→   275→  useEffect(() => {
   276→   276→   276→    const map = mapRef.current;
   277→   277→   277→    if (!map) return;
   278→   278→   278→
   279→   279→   279→    const onLoading = () => {
   280→   280→   280→      if (clearTimeoutRef.current) clearTimeout(clearTimeoutRef.current);
   281→   281→   281→      setError(false);
   282→   282→   282→      setLoading(true);
   283→   283→   283→    };
   284→   284→   284→    const onLoad = () => {
   285→   285→   285→      setLoading(false);
   286→   286→   286→      setError(false);
   287→   287→   287→    };
   288→   288→   288→    const onTileError = () => {
   289→   289→   289→      setLoading(false);
   290→   290→   290→      setError(true);
   291→   291→   291→      clearTimeoutRef.current = setTimeout(() => setError(false), 5000);
   292→   292→   292→    };
   293→   293→   293→
   294→   294→   294→    map.on('tileloadstart', onLoading);
   295→   295→   295→    map.on('tileload', onLoad);
   296→   296→   296→    map.on('load', onLoad);
   297→   297→   297→    map.on('tileerror', onTileError);
   298→   298→   298→
   299→   299→   299→    return () => {
   300→   300→   300→      map.off('tileloadstart', onLoading);
   301→   301→   301→      map.off('tileload', onLoad);
   302→   302→   302→      map.off('load', onLoad);
   303→   303→   303→      map.off('tileerror', onTileError);
   304→   304→   304→      if (clearTimeoutRef.current) clearTimeout(clearTimeoutRef.current);
   305→   305→   305→    };
   306→   306→   306→  }, [mapRef]);
   307→   307→   307→
   308→   308→   308→  return { loading, error };
   309→   309→   309→}
